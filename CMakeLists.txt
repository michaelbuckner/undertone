cmake_minimum_required(VERSION 3.22)
project(Undertone VERSION 0.11.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(UNDERTONE_APP "Build the native desktop application" ON)
add_library(undertone_core src/core/Music.cpp src/core/Engine.cpp src/core/MidiTimeline.cpp src/core/Detail.cpp src/core/Percussion.cpp src/core/Composition.cpp)
target_include_directories(undertone_core PUBLIC src)
if(MSVC)
  target_compile_options(undertone_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(undertone_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(core_tests tests/CoreTests.cpp)
target_link_libraries(core_tests PRIVATE undertone_core)
add_test(NAME core_tests COMMAND core_tests)
add_executable(evolution_tests tests/EvolutionTests.cpp)
target_link_libraries(evolution_tests PRIVATE undertone_core)
add_test(NAME evolution_tests COMMAND evolution_tests)
add_executable(detail_tests tests/DetailTests.cpp)
target_link_libraries(detail_tests PRIVATE undertone_core)
add_test(NAME detail_tests COMMAND detail_tests)
add_executable(melody_tests tests/MelodyTests.cpp)
target_link_libraries(melody_tests PRIVATE undertone_core)
add_test(NAME melody_tests COMMAND melody_tests)
add_executable(composition_tests tests/CompositionTests.cpp)
target_link_libraries(composition_tests PRIVATE undertone_core)
add_test(NAME composition_tests COMMAND composition_tests)
add_executable(cycle_tests tests/CycleTests.cpp)
target_link_libraries(cycle_tests PRIVATE undertone_core)
add_test(NAME cycle_tests COMMAND cycle_tests)
add_executable(refinement_tests tests/RefinementTests.cpp)
target_link_libraries(refinement_tests PRIVATE undertone_core)
add_test(NAME refinement_tests COMMAND refinement_tests)
add_executable(percussion_tests tests/PercussionTests.cpp)
target_link_libraries(percussion_tests PRIVATE undertone_core)
add_test(NAME percussion_tests COMMAND percussion_tests)
add_executable(render_sketch tests/RenderSketch.cpp)
target_link_libraries(render_sketch PRIVATE undertone_core)
add_executable(midi_tests tests/MidiTests.cpp)
target_link_libraries(midi_tests PRIVATE undertone_core)
add_test(NAME midi_tests COMMAND midi_tests)
add_executable(render_demo tests/RenderDemo.cpp)
target_link_libraries(render_demo PRIVATE undertone_core)
add_executable(soak_test tests/Soak.cpp)
target_link_libraries(soak_test PRIVATE undertone_core)
if(UNDERTONE_APP)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/JUCE/CMakeLists.txt")
    add_subdirectory(vendor/JUCE EXCLUDE_FROM_ALL)
  else()
    include(FetchContent)
    FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
      GIT_TAG 3fb5cbc0340758596dac39f7c8de47d562b4896e)
    FetchContent_MakeAvailable(JUCE)
  endif()
  juce_add_gui_app(Undertone PRODUCT_NAME "Undertone" COMPANY_NAME "Mike"
    BUNDLE_ID "local.mike.undertone" NEEDS_CURL FALSE NEEDS_WEB_BROWSER FALSE)
  if(WIN32)
    # JUCE 9.0.4's RC rule omits Info.txt; regenerate Windows version metadata on version changes.
    get_target_property(undertone_generated_dir Undertone JUCE_GENERATED_SOURCES_DIRECTORY)
    get_target_property(undertone_info_file Undertone JUCE_INFO_FILE)
    add_custom_command(OUTPUT "${undertone_generated_dir}/Undertone_resources.rc"
      APPEND DEPENDS "${undertone_info_file}")
  endif()
  target_sources(Undertone PRIVATE src/app/Main.cpp src/app/MainComponent.cpp src/app/InstrumentTheme.cpp src/app/SoundEditor.cpp src/app/SessionBrowser.cpp src/app/SessionLibrary.cpp src/app/LibraryController.cpp src/app/SessionFile.cpp src/app/MidiExport.cpp)
  target_compile_definitions(Undertone PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
    JUCE_APPLICATION_NAME_STRING="Undertone" JUCE_APPLICATION_VERSION_STRING="${PROJECT_VERSION}")
  target_link_libraries(Undertone PRIVATE undertone_core juce::juce_audio_utils
    juce::juce_recommended_config_flags)
  juce_add_console_app(session_tests PRODUCT_NAME "Session tests")
  target_sources(session_tests PRIVATE tests/SessionTests.cpp src/app/SessionFile.cpp)
  target_link_libraries(session_tests PRIVATE undertone_core juce::juce_core)
  target_compile_definitions(session_tests PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
  add_test(NAME session_tests COMMAND session_tests)
  juce_add_console_app(library_tests PRODUCT_NAME "Library tests")
  target_sources(library_tests PRIVATE tests/LibraryTests.cpp src/app/SessionLibrary.cpp src/app/SessionFile.cpp)
  target_link_libraries(library_tests PRIVATE undertone_core juce::juce_core)
  target_compile_definitions(library_tests PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
  add_test(NAME library_tests COMMAND library_tests)
  juce_add_console_app(export_tests PRODUCT_NAME "Export tests")
  target_sources(export_tests PRIVATE tests/ExportTests.cpp src/app/MidiExport.cpp src/app/SessionFile.cpp)
  target_link_libraries(export_tests PRIVATE undertone_core juce::juce_audio_basics)
  target_compile_definitions(export_tests PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
  add_test(NAME export_tests COMMAND export_tests)
  install(TARGETS Undertone RUNTIME DESTINATION . BUNDLE DESTINATION .)
  if(WIN32)
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
    include(InstallRequiredSystemLibraries)
  endif()
  install(FILES README.md DESTINATION .)
  install(FILES docs/melody-0.5.md DESTINATION docs)
  install(FILES docs/composition-0.6.md DESTINATION docs)
  install(FILES docs/cycles-0.7.md DESTINATION docs)
  install(FILES docs/performance-0.8.md DESTINATION docs)
  install(FILES docs/ui-0.9.md DESTINATION docs)
  install(FILES docs/sound-editor-0.10.md DESTINATION docs)
  install(FILES docs/library-0.11.md DESTINATION docs)
  install(FILES docs/quick-start.md docs/build-status.md docs/third-party.md docs/software-prd.md docs/hardware-prd.md docs/mac-verification.md docs/sound-design-0.2.md docs/arrangement-0.3.md docs/percussion-0.4.md DESTINATION docs)
  install(FILES "${JUCE_SOURCE_DIR}/LICENSE.md" DESTINATION licenses RENAME JUCE-LICENSE.md)
endif()
